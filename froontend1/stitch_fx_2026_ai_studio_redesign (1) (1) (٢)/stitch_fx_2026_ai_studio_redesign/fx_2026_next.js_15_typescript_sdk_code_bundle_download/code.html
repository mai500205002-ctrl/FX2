// lib/sdk/client.ts - Type-safe zero-dependency HTTP client
export interface FXClientConfig {
  apiKey: string;
  clusterEndpoint?: string;
  timeout?: number;
  enclaveVerification?: boolean;
}

export class FXCreativeClient {
  private apiKey: string;
  private endpoint: string;

  constructor(config: FXClientConfig) {
    this.apiKey = config.apiKey;
    this.endpoint = config.clusterEndpoint || 'https://cluster.fx2026.internal';
  }

  public readonly pipeline = {
    runBatch: async (payload: any) => {
      const res = await fetch(`${this.endpoint}/v5/creative/batch`, {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${this.apiKey}`,
          'Content-Type': 'application/json',
          'X-FX-Hardware-Attest': 'ed25519-enclave',
        },
        body: JSON.stringify(payload),
      });
      if (!res.ok) throw new Error(`[FX-2026 Pipeline Failed] ${res.statusText}`);
      return await res.json();
    }
  };
}
// app/api/v5/creative/saas-route/route.ts - Next.js 15 App Router Edge Endpoint
import { NextRequest, NextResponse } from 'next/server';
import { FXCreativeClient } from '@/lib/sdk/client';

export const runtime = 'edge'; // Ultra-low latency

export async function POST(req: NextRequest) {
  try {
    const body = await req.json();
    const client = new FXCreativeClient({
      apiKey: process.env.FX_ATLAS_KEY!,
      clusterEndpoint: process.env.FX_CLUSTER_URL,
    });

    const result = await client.pipeline.runBatch(body);
    return NextResponse.json({ success: true, data: result });
  } catch (err: any) {
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}
